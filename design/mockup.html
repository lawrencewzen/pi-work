// Wrap multi-hunk file patch in durable transaction
export async function applyDurablePatch(
  path: string,
  hunks: PatchHunk[],
  ctx: DurableExecutionContext
): Promise<EditResult> {
  const journal = await ctx.openJournal(`patch:${path}`);

  for (const [idx, hunk] of hunks.entries()) {
    if (journal.isCommitted(idx)) continue; // skip applied
    await applyHunkAtomic(path, hunk);
    await ctx.checkpoint({ step: idx, hash: hunk.sha256 });
  }

  return journal.finalize();
}
